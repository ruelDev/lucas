--------------------------CURRENT LOGGING DRIVER-----------------------------
This will log and separate in per day basis

'stack' => [
            'driver' => 'stack',
            'channels' => ['daily_debug', 'daily_info', 'daily_critical', 'daily_error'],
            'ignore_exceptions' => true,
        ],


        'daily_debug' => [
            'driver' => 'single',
            'path' => storage_path('logs/'. date("F Y") . '/'. date("F j, Y") .'/debug.log'),
            'level' => 'debug',
            'replace_placeholders' => true,
        ],

        'daily_info' => [
            'driver' => 'single',
            'path' => storage_path('logs/'. date("F Y") . '/'. date("F j, Y") .'/info/info.log'),
            'level' => 'info',
            'days' => env('LOG_DAILY_DAYS', 7),
            'replace_placeholders' => true,
        ],

        'daily_critical' => [
            'driver' => 'single',
            'path' => storage_path('logs/'. date("F Y") . '/'. date("F j, Y") .'/critical.log'),
            'level' => 'critical',
            'replace_placeholders' => true,
        ],

        'daily_error' => [
            'driver' => 'single',
            'path' => storage_path('logs/'. date("F Y") . '/'. date("F j, Y") .'/error/error.log'),

---------------------------LOGGING EVENTS PER FUNCTIONS---------------------------
During function calls, to log an event, you may follow these syntax to logs

Sample logging 

use Illuminate\Support\Facades\Log;

        Log::info('Accessed the Branch Management Page', ['user_id' => Auth::id()]);

        Log::info('Created a new Branch', ['user_id' => Auth::id(), 'new_data' => json_encode($newData)]);

        Log::info('Updated a Branch', ['user_id' => Auth::id(), 'selected_branch_id' => $branch_management->id, 'old_data' => json_encode($oldData), 'new_data' => json_encode($newData)]);

        Log::info('Deleted a Branch', ['user_id' => Auth::id(), 'deleted_branch_id' => $id]);

        Log::error('Error submitting a request);

        OTHER FORMATS ASIDE FROM LOG::INFO-------------------------

        Log::emergency($message);
        Log::alert($message);
        Log::critical($message);
        Log::error($message);
        Log::warning($message);
        Log::notice($message);
        Log::info($message);
